def campo(parent, label, row, col=0, width=220, placeholder=""):
    ctk.CTkLabel(parent, text=label, anchor="w").grid(
        row=row, column=col, sticky="w", padx=10, pady=(8, 0)
    )
    entry = ctk.CTkEntry(parent, width=width, placeholder_text=placeholder)
    entry.grid(row=row, column=col + 1, sticky="w", padx=10, pady=(8, 0))
    return entry


def botao(parent, label, command, row, col, color="#1f6aa5", width=140):
    btn = ctk.CTkButton(parent, text=label, command=command,
                        fg_color=color, width=width)
    btn.grid(row=row, column=col, padx=10, pady=10, sticky="w")
    return btn


def tabela(parent, colunas, row, col, colspan=4, height=8):
    frame = ctk.CTkFrame(parent)
    frame.grid(row=row, column=col, columnspan=colspan,
               padx=10, pady=10, sticky="nsew")

    style = ttk.Style()
    style.theme_use("clam")
    style.configure("Treeview",
                    background="#2b2b2b",
                    foreground="white",
                    rowheight=26,
                    fieldbackground="#2b2b2b",
                    bordercolor="#2b2b2b",
                    borderwidth=0)
    style.map("Treeview", background=[("selected", "#1f6aa5")])
    style.configure("Treeview.Heading",
                    background="#1a1a2e",
                    foreground="white",
                    relief="flat")

    tree = ttk.Treeview(frame, columns=colunas, show="headings", height=height)
    for col_name in colunas:
        tree.heading(col_name, text=col_name)
        tree.column(col_name, width=130, anchor="center")

    scroll = ttk.Scrollbar(frame, orient="vertical", command=tree.yview)
    tree.configure(yscrollcommand=scroll.set)
    tree.pack(side="left", fill="both", expand=True)
    scroll.pack(side="right", fill="y")
    return tree

class AbaPessoa(ctk.CTkFrame):
    def __init__(self, parent):
        super().__init__(parent)
        self._build()

    def _build(self):
        ctk.CTkLabel(self, text="Cadastro de Pessoa",
                     font=ctk.CTkFont(size=18, weight="bold")).grid(
            row=0, column=0, columnspan=4, pady=(14, 4), sticky="w", padx=10
        )

        self.e_nome = campo(self, "Nome completo", row=1)
        self.e_cpf  = campo(self, "CPF",           row=2, placeholder="000.000.000-00")
        self.e_nasc = campo(self, "Nascimento",    row=3, placeholder="AAAA-MM-DD")
        self.e_tel  = campo(self, "Telefone",      row=4, placeholder="(00) 00000-0000")
        self.e_mail = campo(self, "E-mail",        row=5)

        ctk.CTkLabel(self, text="Modalidades", anchor="w").grid(
            row=6, column=0, sticky="w", padx=10, pady=(8, 0)
        )
        self.frame_modal = ctk.CTkScrollableFrame(self, width=380, height=120)
        self.frame_modal.grid(row=6, column=1, columnspan=3,
                               padx=10, pady=(8, 0), sticky="w")

        self._vars_modal = {}
        for m in listar_modalidades():
            var = ctk.BooleanVar()
            chk = ctk.CTkCheckBox(self.frame_modal, text=m['Nome'], variable=var)
            chk.pack(anchor="w", pady=2)
            self._vars_modal[m['Id']] = var

        botao(self, "Salvar pessoa",  self._salvar, row=7, col=1)
        botao(self, "Listar pessoas", self._listar, row=7, col=2, color="#2d6a4f")

        self.tree = tabela(
            self,
            ["Id", "Nome", "CPF", "Nascimento", "Telefone", "Modalidades"],
            row=8, col=0, colspan=4
        )

    def _salvar(self):
        nome = self.e_nome.get().strip()
        cpf  = self.e_cpf.get().strip()
        nasc = self.e_nasc.get().strip()
        tel  = self.e_tel.get().strip()
        mail = self.e_mail.get().strip()

        if not nome or not cpf or not nasc:
            messagebox.showwarning("Atenção", "Nome, CPF e Nascimento são obrigatórios.")
            return

        existente = buscar_pessoa_por_cpf(cpf)
        if existente:
            messagebox.showerror("CPF duplicado",
                                 f"CPF já cadastrado para: {existente['NomeCompleto']}")
            return

        try:
            pid = cadastrar_pessoa(nome, cpf, nasc, mail or None, tel or None)
            for mid, var in self._vars_modal.items():
                if var.get():
                    vincular_modalidade_pessoa(pid, mid)
            messagebox.showinfo("Sucesso", f"Pessoa '{nome}' cadastrada!")
            self._limpar()
            self._listar()
        except Exception as e:
            messagebox.showerror("Erro", str(e))

    def _listar(self):
        for row in self.tree.get_children():
            self.tree.delete(row)
        for p in listar_pessoas():
            mods = listar_modalidades_da_pessoa(p['Id'])
            nomes_mods = ", ".join(m['Nome'] for m in mods) or "—"
            self.tree.insert("", "end", values=(
                p['Id'], p['NomeCompleto'], p['CPF'],
                p['DataNascimento'], p['Telefone'] or "—", nomes_mods
            ))

    def _limpar(self):
        for e in [self.e_nome, self.e_cpf, self.e_nasc, self.e_tel, self.e_mail]:
            e.delete(0, "end")
        for var in self._vars_modal.values():
            var.set(False)

class AbaTorneio(ctk.CTkFrame):
    def __init__(self, parent):
        super().__init__(parent)
        self._build()

    def _build(self):
        ctk.CTkLabel(self, text="Torneios",
                     font=ctk.CTkFont(size=18, weight="bold")).grid(
            row=0, column=0, columnspan=4, pady=(14, 4), sticky="w", padx=10
        )

        self.e_nome      = campo(self, "Nome",      row=1)
        self.e_local     = campo(self, "Local",     row=2)
        self.e_categoria = campo(self, "Categoria", row=3)
        self.e_inicio    = campo(self, "Início",    row=4, placeholder="AAAA-MM-DD")
        self.e_fim       = campo(self, "Fim",       row=5, placeholder="AAAA-MM-DD")
        self.e_descricao = campo(self, "Descrição", row=6)

        ctk.CTkLabel(self, text="Modalidade", anchor="w").grid(
            row=7, column=0, sticky="w", padx=10, pady=(8, 0)
        )
        self._modal_ids   = []
        self._modal_nomes = []
        for m in listar_modalidades():
            self._modal_ids.append(m['Id'])
            self._modal_nomes.append(m['Nome'])

        self.combo_modal = ctk.CTkComboBox(self, values=self._modal_nomes, width=220)
        self.combo_modal.grid(row=7, column=1, padx=10, pady=(8, 0), sticky="w")

        botao(self, "Salvar torneio",  self._salvar, row=8, col=1)
        botao(self, "Listar torneios", self._listar, row=8, col=2, color="#2d6a4f")

        self.tree = tabela(
            self,
            ["Id", "Nome", "Modalidade", "Local", "Início", "Fim", "Status"],
            row=9, col=0, colspan=4
        )

    def _salvar(self):
        nome      = self.e_nome.get().strip()
        local     = self.e_local.get().strip()
        categ     = self.e_categoria.get().strip()
        inicio    = self.e_inicio.get().strip()
        fim       = self.e_fim.get().strip()
        descricao = self.e_descricao.get().strip()
        modal_nome = self.combo_modal.get()

        if not nome or not modal_nome:
            messagebox.showwarning("Atenção", "Nome e Modalidade são obrigatórios.")
            return

        try:
            modal_id = self._modal_ids[self._modal_nomes.index(modal_nome)]
            criar_torneio(
                nome=nome, modalidade_id=modal_id,
                descricao=descricao or None,
                data_inicio=inicio or None,
                data_fim=fim or None,
                local=local or None,
                categoria=categ or None,
            )
            messagebox.showinfo("Sucesso", f"Torneio '{nome}' criado!")
            self._limpar()
            self._listar()
        except Exception as e:
            messagebox.showerror("Erro", str(e))

    def _listar(self):
        for row in self.tree.get_children():
            self.tree.delete(row)
        for t in listar_torneios():
            self.tree.insert("", "end", values=(
                t['Id'], t['Nome'], t.get('NomeModalidade', '—'),
                t['Local'] or '—', t['DataInicio'] or '—',
                t['DataFim'] or '—', t['Status']
            ))

    def _limpar(self):
        for e in [self.e_nome, self.e_local, self.e_categoria,
                  self.e_inicio, self.e_fim, self.e_descricao]:
            e.delete(0, "end")

class AbaInscricao(ctk.CTkFrame):
    def __init__(self, parent):
        super().__init__(parent)
        self._build()

    def _build(self):
        ctk.CTkLabel(self, text="Inscrição de Time em Torneio",
                     font=ctk.CTkFont(size=18, weight="bold")).grid(
            row=0, column=0, columnspan=4, pady=(14, 4), sticky="w", padx=10
        )

        ctk.CTkLabel(self, text="Torneio", anchor="w").grid(
            row=1, column=0, sticky="w", padx=10, pady=(8, 0)
        )
        self._torneio_ids   = []
        self._torneio_nomes = []
        self.combo_torneio = ctk.CTkComboBox(self, values=[], width=300)
        self.combo_torneio.grid(row=1, column=1, padx=10, pady=(8, 0), sticky="w")

        ctk.CTkLabel(self, text="Time", anchor="w").grid(
            row=2, column=0, sticky="w", padx=10, pady=(8, 0)
        )
        self._time_ids   = []
        self._time_nomes = []
        self.combo_time = ctk.CTkComboBox(self, values=[], width=300)
        self.combo_time.grid(row=2, column=1, padx=10, pady=(8, 0), sticky="w")

        botao(self, "↺ Atualizar",          self._carregar_combos, row=3, col=0, color="#555", width=120)
        botao(self, "Inscrever time",        self._inscrever,       row=3, col=1)
        botao(self, "Ver times do torneio",  self._listar,          row=3, col=2, color="#2d6a4f")

        self.tree = tabela(
            self,
            ["InscricaoId", "Time", "Instituição", "Status"],
            row=4, col=0, colspan=4
        )

        self._carregar_combos()

    def _carregar_combos(self):
        self._torneio_ids, self._torneio_nomes = [], []
        for t in listar_torneios():
            self._torneio_ids.append(t['Id'])
            self._torneio_nomes.append(f"{t['Id']} – {t['Nome']}")
        self.combo_torneio.configure(values=self._torneio_nomes)
        if self._torneio_nomes:
            self.combo_torneio.set(self._torneio_nomes[0])

        self._time_ids, self._time_nomes = [], []
        for t in listar_times():
            self._time_ids.append(t['Id'])
            self._time_nomes.append(f"{t['Id']} – {t['Nome']}")
        self.combo_time.configure(values=self._time_nomes)
        if self._time_nomes:
            self.combo_time.set(self._time_nomes[0])

    def _inscrever(self):
        try:
            tidx = self._torneio_nomes.index(self.combo_torneio.get())
            eidx = self._time_nomes.index(self.combo_time.get())
            iid  = inscrever_time(self._torneio_ids[tidx], self._time_ids[eidx])
            atualizar_status_inscricao(iid, 'confirmada')
            messagebox.showinfo("Sucesso", "Time inscrito e confirmado!")
            self._listar()
        except ValueError as e:
            messagebox.showerror("Erro", "Time já inscrito nesse torneio ou seleção inválida.")
        except Exception as e:
            messagebox.showerror("Erro", str(e))

    def _listar(self):
        for row in self.tree.get_children():
            self.tree.delete(row)
        try:
            tidx = self._torneio_nomes.index(self.combo_torneio.get())
            tid  = self._torneio_ids[tidx]
            for t in listar_times_do_torneio(tid):
                self.tree.insert("", "end", values=(
                    t['InscricaoId'], t['NomeTime'],
                    t['Instituicao'] or '—', t['StatusInscricao']
                ))
        except Exception as e:
            messagebox.showerror("Erro", str(e))

        self.combo_pessoa.configure(values=self._pessoa_nomes)
        if self._pessoa_nomes:
            self.combo_pessoa.set(self._pessoa_nomes[0])

            iidx    = self._insc_nomes.index(self.combo_insc.get())
            pidx    = self._pessoa_nomes.index(self.combo_pessoa.get())
            papel   = self.combo_papel.get()
            camisa  = self.e_camisa.get().strip()
            posicao = self.e_posicao.get().strip()

            adicionar_membro(
                inscricao_time_id=self._insc_ids[iidx],
                pessoa_id=self._pessoa_ids[pidx],
                papel=papel,
                numero_camisa=int(camisa) if camisa else None,
                posicao=posicao or None,
                titular=self.var_titular.get()
            )
            messagebox.showinfo("Sucesso", "Membro adicionado!")
            self._listar()
        except Exception as e:
            messagebox.showerror("Erro", str(e))

    def _listar(self):
        for row in self.tree.get_children():
            self.tree.delete(row)
        try:
            iidx = self._insc_nomes.index(self.combo_insc.get())
            iid  = self._insc_ids[iidx]
            for m in listar_membros_da_inscricao(iid):
                self.tree.insert("", "end", values=(
                    m['MembroId'], m['NomeCompleto'], m['CPF'],
                    m['Papel'], m['NumeroCamisa'] or '—',
                    m['Posicao'] or '—',
                    'Sim' if m['Titular'] else 'Não'
                ))
        except Exception as e:
            messagebox.showerror("Erro", str(e))

def criar_torneio(nome, modalidade_id, descricao=None, data_inicio=None,
                  data_fim=None):
    with get_conn() as conn:
        cursor = conn.execute('''
            INSERT INTO Torneio (Nome, Descricao, DataInicio, DataFim, ModalidadeId)
            VALUES (?, ?, ?, ?, ?)
        ''', (nome, descricao, data_inicio, data_fim, modalidade_id))
        torneio_id = cursor.lastrowid
    print(f"Torneio '{nome}' criado com Id {torneio_id}.")
    return torneio_id
